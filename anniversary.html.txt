/* ══════ FLOATING HEARTS ══════ */
const canvas = document.getElementById('hearts-canvas');
const ctx = canvas.getContext('2d');

function resize() {
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;
}
resize();
window.addEventListener('resize', resize);

class Heart {
  constructor() { this.reset(true); }
  reset(init = false) {
    this.x = Math.random() * canvas.width;
    this.y = init ? Math.random() * canvas.height : canvas.height + 30;
    this.size = 10 + Math.random() * 22;
    this.speed = 0.5 + Math.random() * 1.2;
    this.opacity = 0.15 + Math.random() * 0.55;
    this.drift = (Math.random() - 0.5) * 0.8;
    this.wobble = Math.random() * Math.PI * 2;
    this.wobbleSpeed = 0.02 + Math.random() * 0.03;
    this.hue = 340 + Math.random() * 30; // rose/pink range
  }
  draw() {
    ctx.save();
    ctx.globalAlpha = this.opacity;
    ctx.fillStyle = `hsl(${this.hue}, 80%, 60%)`;
    ctx.shadowColor = `hsl(${this.hue}, 80%, 60%)`;
    ctx.shadowBlur = 12;
    const s = this.size / 10;
    ctx.translate(this.x, this.y);
    ctx.scale(s, s);
    ctx.beginPath();
    ctx.moveTo(0, 0);
    ctx.bezierCurveTo(-5, -5, -10, 0, 0, 7);
    ctx.bezierCurveTo(10, 0, 5, -5, 0, 0);
    ctx.fill();
    ctx.restore();
  }
  update() {
    this.wobble += this.wobbleSpeed;
    this.x += this.drift + Math.sin(this.wobble) * 0.5;
    this.y -= this.speed;
    if (this.y < -40) this.reset();
  }
}

const hearts = Array.from({ length: 55 }, () => new Heart());

function animateHearts() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  hearts.forEach(h => { h.update(); h.draw(); });
  requestAnimationFrame(animateHearts);
}
animateHearts();

const TOTAL_SLIDES = 10;
const SLIDE_DURATION = 1500; // ms per slide → 10 × 1.5s = 15s total

const CAPTIONS = [
  "Our first memory together… 💑",
  "Every moment with you is magic ✨",
  "Six months of beautiful chaos 💕",
  "You make my world so much better 🌹",

let currentSlide = 0;
let slideshowTimer = null;

function buildDots() {
  const dotsEl = document.getElementById('dots');
  dotsEl.innerHTML = '';
  for (let i = 0; i < TOTAL_SLIDES; i++) {
    const d = document.createElement('div');
    d.className = 'dot' + (i === 0 ? ' active' : '');
    dotsEl.appendChild(d);
  }
}

function showSlide(idx) {
  // Hide all placeholders / images
  for (let i = 0; i < TOTAL_SLIDES; i++) {
    const ph = document.getElementById('ph-' + i);
    if (ph) ph.classList.remove('visible');
    const img = document.querySelectorAll('.slide-img')[i];
    if (img) img.classList.remove('visible');
  }

  // Show current
  const ph = document.getElementById('ph-' + idx);
  if (ph) ph.classList.add('visible');
  const img = document.querySelectorAll('.slide-img')[idx];
  if (img) img.classList.add('visible');

  // Update dots
  document.querySelectorAll('.dot').forEach((d, i) => {
    d.classList.toggle('active', i === idx);
  });

  // Caption
  const cap = document.getElementById('caption');
  cap.style.opacity = 0;
  setTimeout(() => {
    cap.textContent = CAPTIONS[idx] || '';
    cap.style.opacity = 1;
  }, 300);
}

function startSlideshow() {
  buildDots();
  showSlide(0);
  currentSlide = 0;

  slideshowTimer = setInterval(() => {
    currentSlide++;
    if (currentSlide >= TOTAL_SLIDES) {
      clearInterval(slideshowTimer);
      // Go to final page after a brief pause
      setTimeout(() => goToPage3(), 1200);
      return;
    }
    showSlide(currentSlide);
  }, SLIDE_DURATION);
}

function startTypewriter() {
  const el = document.getElementById('typewriter-text');
  const cursor = document.getElementById('cursor');
  el.innerHTML = '';
  el.appendChild(cursor);

  let charIdx = 0;
  const chars = [...FULL_MESSAGE]; // unicode-safe split

  function type() {
    if (charIdx >= chars.length) {
      cursor.classList.add('done');
      document.getElementById('p3Hearts').style.display = 'block';
      document.getElementById('p3Sign').style.display = 'block';
      return;
    }

    const ch = chars[charIdx++];
    const textNode = document.createTextNode(ch === '\n' ? '' : ch);

    if (ch === '\n') {
      el.insertBefore(document.createElement('br'), cursor);
      el.insertBefore(document.createElement('br'), cursor);
    } else {
      el.insertBefore(textNode, cursor);
    }

    // Vary typing speed for natural feel
    const delay = ch === '.' || ch === ',' ? 80 : ch === ' ' ? 35 : 28 + Math.random() * 20;
    setTimeout(type, delay);
  }

  type();
}
</script>
